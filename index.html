Number(item.PRDUCT_PRICE).toLocaleString() : '0'} `; tbody.innerHTML += row; }); loading.style.display = 'none'; table.style.display = 'table'; } else { // 결과 코드가 00이 아닌 경우 (인증키 문제 등) const errMsg = data.response ? data.response.header.resultMsg : '알 수 없는 에러'; throw new Error(errMsg); } } catch (error) { loading.style.display = 'none'; console.error("상세 에러 내용:", error); alert(`데이터 로드 실패!\n이유: ${error.message}\n\n※ 재발급 후 24시간이 안 지났다면 좀 더 기다려야 할 수 있습니다.`); } }
